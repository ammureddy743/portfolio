import React, { useState, useMemo } from 'react';

interface TaskItem {
  id: number;
  text: string;
  tag: 'frontend' | 'mobile' | 'servicenow';
  completed: boolean;
}

export const TeamConnectTaskManager: React.FC = () => {
  // Component state
  const [tasks, setTasks] = useState<TaskItem[]>([
    { id: 1, text: 'Architect reusable Button & Modal UI components', tag: 'frontend', completed: true },
    { id: 2, text: 'Optimize FlatList render performance in React Native', tag: 'mobile', completed: true }
  ]);
  const [filter, setFilter] = useState<string>('all');

  const filteredTasks = useMemo(() => {
    return filter === 'all' ? tasks : tasks.filter(t => t.tag === filter);
  }, [tasks, filter]);

  return (
    <div className="p-6 rounded-2xl bg-zinc-900 border border-zinc-800">
      {filteredTasks.map(task => (
        <TaskRow key={task.id} task={task} />
      ))}
    </div>
  );
};
