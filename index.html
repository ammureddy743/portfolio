import React, { useState, useMemo } from 'react';

interface TaskItem {
  id: number;
  text: string;
  tag: 'frontend' | 'mobile' | 'servicenow';
  completed: boolean;
}

export const TeamConnectTaskManager: React.FC = () => {
  // Local component state for tasks and active filter
  const [tasks, setTasks] = useState<TaskItem[]>([
    { id: 1, text: 'Architect reusable Button & Modal UI components', tag: 'frontend', completed: true },
    { id: 2, text: 'Optimize FlatList render performance in React Native', tag: 'mobile', completed: true },
    { id: 3, text: 'Configure ServiceNow automated catalog request workflow', tag: 'servicenow', completed: false },
  ]);
  const [filter, setFilter] = useState<string>('all');

  // Memoized list calculation preventing redundant renders
  const filteredTasks = useMemo(() => {
    return filter === 'all' 
      ? tasks 
      : tasks.filter(t => t.tag === filter);
  }, [tasks, filter]);

  const toggleTask = (id: number) => {
    setTasks(prev => prev.map(t => t.id === id ? { ...t, completed: !t.completed } : t));
  };

  return (
    <div className="p-6 rounded-2xl bg-slate-900 border border-slate-800">
      {/* Filter Tabs, Task Item Rows, & Add Form */}
      {filteredTasks.map(task => (
        <TaskRow 
          key={task.id} 
          task={task} 
          onToggle={() => toggleTask(task.id)} 
        />
      ))}
    </div>
  );
};
